# Pannes : page d'erreur et alertes — plan de construction

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** chaque panne du site (serveur, navigateur) est enregistrée une fois, l'administrateur est prévenu sur son téléphone et par e-mail, et l'utilisateur voit une page d'erreur en français.

**Architecture:** une table `app_errors` (une ligne par panne, reconnue par une empreinte) ; `reportError` (service) écrit la ligne et décide d'alerter ; trois portes d'entrée : `onRequestError` (instrumentation Next.js), `runAction` (actions), `POST /api/erreurs` (navigateur). Alertes : `sendPush` existant vers les abonnements des super admins + e-mail Gmail (`sendMail`). Page Admin › Pannes avec la carte « Alertes sur ce téléphone ».

**Tech Stack:** Next.js 16.4 (App Router, `error.tsx` avec `retry`, `global-error.tsx`, `instrumentation.ts`), Drizzle + Postgres (Neon), web-push, nodemailer, Vitest.

**Spec:** `docs/superpowers/specs/2026-10-28-sauvegardes-et-pannes-design.md` (section « Pannes »).

## Global Constraints

- Tous les textes visibles en français simple ; pas de jargon.
- Migration **ajout seulement** : `drizzle/0012_pannes.sql` (accord de l'utilisateur donné le 2026-10-28).

```ts
/** Pannes du site : reconnaître une même panne et décider quand prévenir l'administrateur. */

export const ERROR_SOURCES = ["serveur", "navigateur", "sauvegarde"] as const;
export type ErrorSource = (typeof ERROR_SOURCES)[number];

const UUID = /[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/gi;

/** Adresse sans paramètres ni identifiants : « /ventes/<id>?x=1 » → « /ventes/:id ». */
export function normalizeRoute(route: string): string {
  const path = route.split("?")[0] || "/";
  return path.replace(UUID, ":id").slice(0, 200);
}

/** Une même panne revient avec d'autres nombres ou identifiants : on les remplace pour la reconnaître. */
export function normalizeErrorMessage(message: string): string {
  return message.replace(UUID, ":id").replace(/\d+/g, "#").replace(/\s+/g, " ").trim().slice(0, 300);
}

/** Clé d'une panne ; le serveur en garde l'empreinte SHA-1. */
export function errorKey(source: ErrorSource, route: string, message: string): string {
  return `${source}|${normalizeRoute(route)}|${normalizeErrorMessage(message)}`;
}

export const REALERT_AFTER_HOURS = 24;

/** Prévenir si la panne est nouvelle, revenue après « Réglée », ou revenue après 24 h de calme. */
export function shouldAlert(previous: { lastSeenAt: Date; resolvedAt: Date | null } | null, now: Date): boolean {
  if (!previous) return true;
  if (previous.resolvedAt) return true;
  return now.getTime() - previous.lastSeenAt.getTime() >= REALERT_AFTER_HOURS * 3_600_000;
}

/** Redirection, page introuvable… : le fonctionnement normal de Next.js, pas une panne. */
export function isFrameworkSignal(digest: string | undefined): boolean {
  return Boolean(digest && /^(NEXT_REDIRECT|NEXT_NOT_FOUND|NEXT_HTTP_ERROR_FALLBACK|DYNAMIC_SERVER_USAGE|BAILOUT_TO_CLIENT_SIDE_RENDERING)/.test(digest));
}

export function errorDigest(error: unknown): string | undefined {
  if (typeof error !== "object" || error === null || !("digest" in error)) return undefined;
  const digest = (error as { digest?: unknown }).digest;
  return typeof digest === "string" && digest ? digest : undefined;
}

export function describeError(error: unknown): string {
  if (error instanceof Error) return error.message ? `${error.name}: ${error.message}` : error.name;
  return typeof error === "string" && error ? error : "Erreur inconnue";
}

/** Code court affiché sur la page d'erreur pour retrouver la panne. */
export function shortCode(digest: string | null | undefined): string | null {
  const clean = (digest ?? "").replace(/[^a-z0-9]/gi, "");
  return clean ? clean.slice(0, 6).toUpperCase() : null;
}

const SOURCE_LABELS: Record<ErrorSource, string> = {
  serveur: "serveur",
  navigateur: "téléphone ou ordinateur",
  sauvegarde: "sauvegarde",
};

export function alertText(input: { source: ErrorSource; route: string; message: string; count: number }): { title: string; body: string } {
  const where = input.source === "sauvegarde" ? "Sauvegarde" : input.route;
  const times = input.count <= 1 ? "1re fois" : `${input.count} fois`;
  return { title: `Panne sur Sama Stock — ${where}`, body: `${SOURCE_LABELS[input.source]} : ${input.message.slice(0, 140)} (${times})` };
}
```

```ts
import "server-only";
import { createHash } from "node:crypto";
import { eq, inArray, sql } from "drizzle-orm";
import { headers } from "next/headers";
import { db } from "@/db";
import { appErrors, pushSubscriptions, user } from "@/db/schema";
import { alertText, describeError, errorDigest, errorKey, isFrameworkSignal, normalizeRoute, shouldAlert, type ErrorSource } from "@/lib/domain/app-errors";
import { LEGAL } from "@/lib/legal";
import { ActionError } from "../action";
import { adminEmails } from "../admin-emails";
import { alertEmail, isMailEnabled, sendMail } from "../mail";
import { sendPush } from "../push";

export interface ErrorReport {
  source: ErrorSource;
  route: string;
  message: string;
  digest?: string | null;
}

/** En ligne seulement ; l'ordinateur de développement partage la base de production (sauf ERRORS_DEV=1 pour les essais). */
function recording(): boolean {
  return process.env.VERCEL_ENV === "production" || process.env.ERRORS_DEV === "1";
}

const siteUrl = () => (process.env.NEXT_PUBLIC_APP_URL || "https://sama-stock-one.vercel.app").replace(/\/$/, "");

/** Base injoignable : e-mail seul, au plus un par panne et par heure sur ce serveur. */
const mailedWithoutDb = new Map<string, number>();
/** Pannes envoyées par les navigateurs (adresse publique) : au plus une alerte toutes les 5 minutes par serveur. */
let lastBrowserAlert = 0;

/** Enregistre une panne et prévient l'administrateur si besoin. Ne lève jamais d'erreur. */
export async function reportError(report: ErrorReport): Promise<void> {
  const message = (report.message || "Erreur sans message").slice(0, 500);
  const route = normalizeRoute(report.route || "/");
  console.error("[panne]", report.source, route, message);
  if (!recording()) return;
  const fingerprint = createHash("sha1").update(errorKey(report.source, route, message)).digest("hex");
  const now = new Date();
  let alertCount: number | null;
  try {
    alertCount = await db.transaction(async (tx) => {
      const inserted = await tx
        .insert(appErrors)
        .values({ fingerprint, source: report.source, route, message, digest: report.digest ?? null, alertedAt: now })
        .onConflictDoNothing({ target: appErrors.fingerprint })
        .returning({ id: appErrors.id });
      if (inserted.length > 0) return 1;
      const [previous] = await tx.select().from(appErrors).where(eq(appErrors.fingerprint, fingerprint)).for("update");
      if (!previous) return null;
      const notify = shouldAlert({ lastSeenAt: previous.lastSeenAt, resolvedAt: previous.resolvedAt }, now);
      const count = previous.count + 1;
      await tx
        .update(appErrors)
        .set({ count, message, digest: report.digest ?? previous.digest, lastSeenAt: now, resolvedAt: null, ...(notify ? { alertedAt: now } : {}) })
        .where(eq(appErrors.id, previous.id));
      return notify ? count : null;
    });
  } catch (dbError) {
    console.error("[panne] base injoignable", dbError);
    const last = mailedWithoutDb.get(fingerprint) ?? 0;
    if (now.getTime() - last < 3_600_000) return;
    mailedWithoutDb.set(fingerprint, now.getTime());
    await mailAdmin(alertText({ source: report.source, route, message, count: 1 }));
    return;
  }
  if (alertCount === null) return;
  if (report.source === "navigateur") {
    if (now.getTime() - lastBrowserAlert < 5 * 60_000) return;
    lastBrowserAlert = now.getTime();
  }
  await alertAdmins(alertText({ source: report.source, route, message, count: alertCount }));
}

/** Panne vue par Next.js (page, route, action non attrapée). */
export async function reportServerError(error: unknown, path: string): Promise<void> {
  const digest = errorDigest(error);
  if (isFrameworkSignal(digest) || error instanceof ActionError) return;
  await reportError({ source: "serveur", route: path, message: describeError(error), digest });
}

/** Page d'où vient une action (en-tête Referer). */
export async function currentRoute(): Promise<string> {
  try {
    const referer = (await headers()).get("referer");
    return referer ? new URL(referer).pathname : "/";
  } catch {
    return "/";
  }
}

/** Téléphones des super admins (ADMIN_EMAILS). */
export async function adminPushTargets() {
  const emails = adminEmails();
  if (emails.length === 0) return [];
  return db
    .select({ id: pushSubscriptions.id, endpoint: pushSubscriptions.endpoint, p256dh: pushSubscriptions.p256dh, auth: pushSubscriptions.auth })
    .from(pushSubscriptions)
    .innerJoin(user, eq(user.id, pushSubscriptions.userId))
    .where(inArray(sql`lower(${user.email})`, emails));
}

async function mailAdmin(text: { title: string; body: string }, path = "/admin/pannes"): Promise<boolean> {
  if (!isMailEnabled) return false;
  try {
    await sendMail({ to: LEGAL.email, ...alertEmail({ ...text, url: `${siteUrl()}${path}` }) });
    return true;
  } catch (error) {
    console.error("[panne] e-mail impossible", error);
    return false;
  }
}

/** Prévient les super admins : notification sur leurs téléphones et e-mail. Ne lève jamais d'erreur. */
export async function alertAdmins(text: { title: string; body: string }, path = "/admin/pannes"): Promise<{ phones: number; mail: boolean }> {
  let phones = 0;
  try {
    phones = await sendPush(
      await adminPushTargets(),
      { id: crypto.randomUUID(), kind: "panne", title: text.title, body: text.body, url: path, icon: "/icon-192.png", tag: `panne-${Date.now()}` },
      "high",
    );
  } catch (error) {
    console.error("[panne] notification impossible", error);
  }
  const mail = await mailAdmin(text, path);
  return { phones, mail };
}
```

```ts
import { after, type NextRequest } from "next/server";
import { z } from "zod";
import { reportError } from "@/server/services/app-errors";

const reportSchema = z.object({
  message: z.string().min(1).max(500),
  path: z.string().max(300),
});

/** Panne vue dans le navigateur (page d'erreur affichée). Publique : la page d'erreur peut s'afficher sans connexion. */
export async function POST(request: NextRequest) {
  const raw = await request.text();
  if (raw.length > 2048) return new Response(null, { status: 413 });
  let body: unknown;
  try {
    body = JSON.parse(raw);
  } catch {
    return new Response(null, { status: 400 });
  }
  const parsed = reportSchema.safeParse(body);
  if (!parsed.success) return new Response(null, { status: 400 });
  after(() => reportError({ source: "navigateur", route: parsed.data.path, message: parsed.data.message }));
  return new Response(null, { status: 204 });
}
```

```tsx
function subscribeOnline(onChange: () => void) {
  window.addEventListener("online", onChange);
  window.addEventListener("offline", onChange);
  return () => {
    window.removeEventListener("online", onChange);
    window.removeEventListener("offline", onChange);
  };
}

/** « Un problème est survenu » : réessayer, revenir à l'accueil, ou nous écrire avec le code de la panne. */
export function ErrorScreen({
  error,
  retry,
  homeHref = "/",
  fullPage = false,
}: {
  error: unknown;
  retry: () => void;
  homeHref?: string;
  fullPage?: boolean;
}) {
  const pathname = usePathname() ?? "/";
  const offline = useSyncExternalStore(subscribeOnline, () => !navigator.onLine, () => false);
  const digest = errorDigest(error);
  const code = shortCode(digest);

  const help = salesLink(`Bonjour Sama Stock, j'ai un problème sur l'application${code ? ` (code ${code})` : ""}, page ${pathname}.`);

      <p className="mt-3 text-muted-foreground">

          ? "Vérifiez la connexion du téléphone, puis réessayez."
          : "Ce n'est pas de votre faute : l'équipe Sama Stock est prévenue. Réessayez dans un instant."}
      </p>
      <div className="mt-8 flex flex-wrap justify-center gap-3">

          type="button"
          onClick={() => retry()}

        >
          <RotateCcw className="size-4" /> Réessayer

          <Logo />
        </Link>

    </div>
  );
}
```

- [ ] **Step 3: `src/server/actions/admin-alerts.ts`**

```ts
"use server";

import { and, eq } from "drizzle-orm";
import { revalidatePath } from "next/cache";
import { headers } from "next/headers";
import { z } from "zod";
import { db } from "@/db";
import { appErrors, pushSubscriptions } from "@/db/schema";
import { pushSubscriptionSchema, type PushSubscriptionInput } from "@/lib/push-subscription";
import { parse, runAction } from "../action";
import { requireSuperAdminAction } from "../context";
import { alertAdmins } from "../services/app-errors";

/** Ce téléphone recevra les alertes de panne de l'administrateur, même application fermée. */
export async function saveAdminPushSubscription(input: PushSubscriptionInput) {
  return runAction(async () => {
    const ctx = await requireSuperAdminAction();
    const sub = parse(pushSubscriptionSchema, input);
    const userAgent = (await headers()).get("user-agent")?.slice(0, 300) ?? null;
    await db
      .insert(pushSubscriptions)
      .values({ userId: ctx.user.id, endpoint: sub.endpoint, p256dh: sub.keys.p256dh, auth: sub.keys.auth, userAgent })
      .onConflictDoUpdate({ target: pushSubscriptions.endpoint, set: { userId: ctx.user.id, p256dh: sub.keys.p256dh, auth: sub.keys.auth, userAgent } });
    return null;
  });
}

export async function removeAdminPushSubscription(endpoint: string) {
  return runAction(async () => {
    const ctx = await requireSuperAdminAction();
    await db.delete(pushSubscriptions).where(and(eq(pushSubscriptions.endpoint, endpoint), eq(pushSubscriptions.userId, ctx.user.id)));
    return null;
  });
}

/** Alerte d'essai : téléphones des administrateurs et e-mail. */
export async function sendAdminTestAlert() {
  return runAction(async () => {
    await requireSuperAdminAction();
    const res = await alertAdmins({ title: "Alertes de panne activées", body: "Vous serez prévenu ici quand Sama Stock a un problème." });
    return { sent: res.phones, mail: res.mail };
  });
}

/** La panne est corrigée ; si elle revient, une nouvelle alerte partira. */
export async function resolveAppError(id: string) {
  return runAction(async () => {
    await requireSuperAdminAction();
    const errorId = parse(z.string().min(1).max(64), id);
    await db.update(appErrors).set({ resolvedAt: new Date() }).where(eq(appErrors.id, errorId));
    revalidatePath("/admin/pannes");
    return null;
  });
}
```

- [ ] **Step 4: `src/server/queries/app-errors.ts`**

```ts
import "server-only";
import { count, desc, isNull, sql } from "drizzle-orm";
import { db } from "@/db";
import { appErrors } from "@/db/schema";

/** Pannes non réglées d'abord, puis les plus récentes. */
export async function listAppErrors(limit = 200) {
  return db
    .select()
    .from(appErrors)
    .orderBy(sql`${appErrors.resolvedAt} is null desc`, desc(appErrors.lastSeenAt))
    .limit(limit);
}

export async function openAppErrorCount(): Promise<number> {
  const [row] = await db.select({ n: count() }).from(appErrors).where(isNull(appErrors.resolvedAt));
  return row?.n ?? 0;
}
```

```tsx
import { CheckCircle2, Globe, Server, DatabaseBackup } from "lucide-react";
import { useTransition } from "react";

export interface AppErrorRow {
  id: string;
  source: string;
  route: string;
  message: string;
  count: number;
  firstSeenAt: string;
  lastSeenAt: string;
  resolvedAt: string | null;
}

const SOURCE = {
  serveur: { label: "Serveur", icon: Server },
  navigateur: { label: "Téléphone ou ordinateur", icon: Globe },
  sauvegarde: { label: "Sauvegarde", icon: DatabaseBackup },

export function AppErrorList({ errors }: { errors: AppErrorRow[] }) {

  if (errors.length === 0) {

      </p>
    );
  }
  return (
    <ul className="space-y-3">
      {errors.map((e) => {
        const source = SOURCE[e.source as keyof typeof SOURCE] ?? SOURCE.serveur;
        return (
          <li key={e.id} className={`rounded-xl border p-4 ${e.resolvedAt ? "opacity-60" : "border-destructive/40 bg-destructive/5"}`}>
            <div className="flex flex-wrap items-start justify-between gap-3">

                  {e.resolvedAt ? ` · réglée ${formatDateTime(e.resolvedAt)}` : ""}
```
